package "SkillManager Family" {

    class Skill {
        + skill_id : str
        + skill_name : str
        + skill_title : str
        + skill_description : str
        + folder_name : str
        + instruction_text : str
        + ragmem_title : str
        + ragmem_description : str
        + ragmem_q : str
        + ragmem_a : str
        + ragmem_record_id : str
        + ragmem_recall_key : str
        + skill_status : str
        + notes : List
        + validate() : List[str]
    }

    class SkillManager {
        + create_or_replace_skill(
            skill:Skill,
            archive_ids:List[str],
            related_ids:List[str]
          ) : Result

        + retrieve_candidates(
            query:str
          ) : List

        + load_selected_skills(
            skill_ids:List[str]
          ) : List[str]
    }

    class SkillCreator {
        + create_skill(
            skill:Skill
          ) : Path

        + update_skill_metadata(
            skill_id:str,
            status:str,
            notes:List
          ) : None
    }

    class SkillRetrieval {
        + search(
            query:str,
            k:int=10,
            threshold:float=0.2
          ) : List

        + load_markdown(
            skill_ids:List[str]
          ) : List[str]
    }
}
